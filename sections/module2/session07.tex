  \vocabpair{Father / Dad}{Padre / Papá}
  \vocabpair{Mother / Mom}{Madre / Mamá}
  \vocabpair{Brother}{Hermano}
  \vocabpair{Sister}{Hermana}
  \vocabpair{Grandma}{Abuela}
  \vocabpair{Grandpa}{Abuelo}
  \vocabpair{Uncle}{Tío}
  \vocabpair{Aunt}{Tía}
  \vocabpair{Cousin}{Primo / Prima}
  \vocabpair{Baby}{Bebé}
\end{vocabcard}

\begin{grammartip}{Introducing Family}
  Use \textbf{This is my...} to introduce someone:\\
  \emph{This is my brother.}\\
  Use \textbf{My [member] is [name]...} to say their name:\\
  \emph{My mom is Maria.}
\end{grammartip}

  \textbf{2.} My mother's name is \answerline[4cm] \\
  \textbf{3.} My \answerline[2cm] is \answerline[4cm] \\
  \textbf{4.} My \answerline[2cm] is \answerline[4cm]
\end{activity}

\begin{activity}{Activity 2: Family Riddles}
  Who are they? Write the family member! (¿Quiénes son? ¡Escribe el miembro de la familia!)\\

  \textbf{2.} My dad's brother is my \answerline[3cm] \\
  \textbf{3.} My mom's baby is my \answerline[3cm]
\end{activity}
