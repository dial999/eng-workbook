\subsection*{How do we look?}

  \vocabpair{Blue eyes}{Ojos azules}
  \vocabpair{Brown eyes}{Ojos marrones}
  \vocabpair{Glasses}{Gafas / Lentes}
  \vocabpair{Young}{Joven}
  \vocabpair{Old}{Viejo / Mayor}
\end{vocabcard}

\begin{grammartip}{IS and HAS GOT}
  Use \textbf{IS} for height and age:\\
  \emph{My dad is tall.} / \emph{My grandma is old.}\\
  Use \textbf{HAS GOT} for hair, eyes, and glasses:\\
  \emph{My sister has got long hair.} / \emph{He has got glasses.}
\end{grammartip}

\begin{activity}{Activity 1: Describe a Family Member}
  Write 4 sentences about a family member. Use IS (2 times) and HAS GOT (2 times). (Escribe 4 oraciones sobre un miembro de tu familia. Usa IS y HAS GOT.)
  \writinglines[4]
\end{activity}

\begin{activity}{Activity 2: True or False?}
  Read and circle True or False about yourself! (¡Lee y encierra en un círculo Verdadero o Falso sobre ti!)\\
  \textbf{1.} I am tall. \hfill \textbf{TRUE \ / \ FALSE}\\
  \textbf{2.} I have got short hair. \hfill \textbf{TRUE \ / \ FALSE}\\
  \textbf{3.} I have got brown eyes. \hfill \textbf{TRUE \ / \ FALSE}\\
  \textbf{4.} I have got glasses. \hfill \textbf{TRUE \ / \ FALSE}\\
  \textbf{5.} I am young. \hfill \textbf{TRUE \ / \ FALSE}
\end{activity}

\begin{activity}{Activity 3: My Best Friend}
  Describe your best friend in 3 sentences! (¡Describe a tu mejor amigo(a) en 3 oraciones!)
  \writinglines[3]
\end{activity}
