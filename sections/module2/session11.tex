\begin{grammartip}{CAN and CAN'T}
  \textbf{CAN} (puede): \emph{A bird can fly.}\\
  \textbf{CAN'T} (no puede): \emph{An elephant can't fly.}\\
  \textbf{Question:} \emph{Can a fish swim? Yes, it can! / No, it can't.}
\end{grammartip}

\begin{activity}{Activity 1: Circle CAN or CAN'T}
  Read the sentences and circle the correct word! (¡Lee las oraciones y encierra la palabra correcta!)\\
  \textbf{1.} A monkey \textbf{CAN / CAN'T} climb trees.\\
  \textbf{2.} A snake \textbf{CAN / CAN'T} run.\\
  \textbf{3.} A parrot \textbf{CAN / CAN'T} talk.\\
  \textbf{4.} A fish \textbf{CAN / CAN'T} fly.\\
  \textbf{5.} A lion \textbf{CAN / CAN'T} roar.\\
  \textbf{6.} A turtle \textbf{CAN / CAN'T} run fast.
\end{activity}
