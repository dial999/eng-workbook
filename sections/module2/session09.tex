\begin{activity}{Activity 2: About Your Pet}
  Answer these questions in full sentences! If you don't have a pet, imagine one. (¡Responde en oraciones completas! Si no tienes mascota, imagina una.)\\
  \textbf{1.} Do you have a pet?\\
  \writinglines[1]
  \textbf{2.} What is its name?\\
  \writinglines[1]
  \textbf{3.} What color is it?\\
  \writinglines[1]
\end{activity}

\begin{activity}{Activity 3: Animal Unscramble}

  \textbf{3.} RTUETL = \answerline[3cm]\\
  \textbf{4.} RRATOP = \answerline[3cm]
\end{activity}
