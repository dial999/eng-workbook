\section*{Session 12: Project 2 - Family \& Pets Poster}
\subsection*{Show and Tell!}

\begin{activity}{1. Poster Planning}
  Plan your poster! Draw your family and your pet, then write your script. (¡Planea tu póster! Dibuja a tu familia y a tu mascota, luego escribe tu guion.)\\[1em]
  \textbf{My Family:} \hspace{5cm} \textbf{My Pet (or Favorite Animal):}\\[0.5em]
  \framebox[6cm]{\rule{0pt}{5cm}} \hfill
  \framebox[6cm]{\rule{0pt}{5cm}}\\[1em]
  
  \textbf{My Script:}
  \writinglines[5]
\end{activity}

\begin{grammartip}{2. Oral Presentation Checklist}
  Check (\faIcon{check}) when you are ready to present!\\
  {[ \ ]} I know my words in English.\\
  {[ \ ]} I have a drawing of my family.\\
  {[ \ ]} I have a drawing of my pet.\\
  {[ \ ]} I can speak loud and clear.
\end{grammartip}

\begin{activity}{3. Peer Questions}
  Ask your classmates questions about their presentation! (¡Hazle preguntas a tus compañeros sobre su presentación!)\\
  \emph{What is your pet's name?}\\
  \emph{How many brothers do you have?}\\
  \emph{Can your pet jump?}
\end{activity}

  Write one word for each! (¡Escribe una palabra para cada uno!)\\

  \textbf{3.} A domestic animal: \answerline[3cm]\\
  \textbf{4.} A wild animal: \answerline[3cm]\\
  \textbf{5.} An action verb: \answerline[3cm]
\end{activity}
