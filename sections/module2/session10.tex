\begin{activity}{Activity 1: Animal Riddles}
  Read and solve the riddles! Which animal is it? (¡Lee y resuelve las adivinanzas! ¿Qué animal es?)\\
  \textbf{1.} It is big and grey, it has a long nose. It is an \answerline[3cm] \\
  \textbf{2.} It is tall, it has a long neck, and it is yellow and brown. It is a \answerline[3cm] \\
  \textbf{3.} It has no legs and it is very long. It is a \answerline[3cm] \\
  \textbf{4.} It is black and white, it looks like a horse. It is a \answerline[3cm]
\end{activity}

\begin{activity}{Activity 2: Animal Habitats}
  Match the animal to its home! Draw a line. (¡Une el animal con su hogar! Dibuja una línea.)\\[1em]
  \textbf{1.} Lion \hspace{3cm} \textbf{A.} Ocean\\
  \textbf{2.} Penguin \hspace{2.4cm} \textbf{B.} Jungle\\
  \textbf{3.} Monkey \hspace{2.5cm} \textbf{C.} Savanna
\end{activity}

\begin{activity}{Activity 3: Write a Riddle}
  Write your own riddle about a wild animal for a classmate! (¡Escribe tu propia adivinanza sobre un animal salvaje para un compañero!)\\
  \emph{It is... It has got... It is a...}\\
  \writinglines[3]
\end{activity}
