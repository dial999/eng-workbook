\section*{Session 20: Weather \& Seasons}
\textbf{Objective:} Relate weather to clothing.

\subsection*{Vocabulary}
\begin{vocabcard}{Weather \& Seasons (El clima y las estaciones)}
  \vocabpair{Sunny}{Soleado}
  \vocabpair{Rainy}{Lluvioso}
  \vocabpair{Cold}{Frío}
  \vocabpair{Hot}{Caluroso}
  \vocabpair{Windy}{Ventoso}
  \vocabpair{Cloudy}{Nublado}
  \vocabpair{Snowy}{Nevado}
  \vocabpair{Stormy}{Tormentoso}
  \vocabpair{Spring}{Primavera}
  \vocabpair{Summer}{Verano}
  \vocabpair{Autumn}{Otoño}
  \vocabpair{Winter}{Invierno}
\end{vocabcard}

\subsection*{Grammar}
\begin{grammartip}{How's the weather today?}
  To ask about the weather, we say: \textbf{How's the weather today?}\\
  \textbf{Answer:}\\
  It is sunny. (Está soleado.)\\
  \textbf{Connecting weather and clothes:}\\
  When it is cold, I wear a jacket. (Cuando hace frío, me pongo una chamarra.)
\end{grammartip}

\subsection*{Activities}

\begin{activity}{Activity 1: What do you wear?}
  Complete the sentences with clothes (Completa las oraciones con ropa).
  
  1. When it is RAINY, I wear \answerline[4cm].\\
  2. When it is SUNNY, I wear \answerline[4cm].\\
  3. When it is COLD, I wear \answerline[4cm].\\
  4. When it is SNOWY, I wear \answerline[4cm].
\end{activity}

\begin{activity}{Activity 2: Seasons Match \faIcon{pencil-alt}}
  Match each season to an activity (Une cada estación con una actividad).
  
  \textbf{Spring} \hspace{2cm} go to the beach\\
  \textbf{Summer} \hspace{2cm} play in the snow\\
  \textbf{Autumn} \hspace{2cm} look at flowers\\
  \textbf{Winter} \hspace{2cm} watch leaves fall
\end{activity}

\begin{activity}{Activity 3: Weather Reporter \faIcon{music}}
  Look outside: what's the weather today? (Mira afuera: ¿cómo está el clima hoy?). Write 3 sentences about the weather and what you wear.
  \writinglines[3]
\end{activity}
