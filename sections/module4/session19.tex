\section*{Session 19: Clothes \& Accessories}
\textbf{Objective:} Identify clothing items.

\subsection*{Grammar}
\begin{grammartip}{What are you wearing?}
  We use the verb \textbf{wearing} (usando / vistiendo) to talk about clothes.\\
  \textbf{Example:}\\
  I am wearing a blue T-shirt. (Llevo puesta una camiseta azul.)\\
  She is wearing red shoes. (Ella lleva puestos zapatos rojos.)
\end{grammartip}

\subsection*{Activities}

\begin{activity}{Activity 1: What are you wearing?}
  Write 3 sentences about what YOU are wearing right now (Escribe 3 oraciones sobre lo que llevas puesto ahora mismo).
  \writinglines[3]
\end{activity}

\begin{activity}{Activity 2: Fashion Police \faIcon{lightbulb}}
  Describe a famous cartoon character's outfit (Describe la ropa de un personaje famoso de caricatura). Write 3 sentences.
  \writinglines[3]
\end{activity}

\begin{activity}{Activity 3: True or False \faIcon{check}}
  Read the sentences and write T (True) or F (False) (Lee las oraciones y escribe T para Verdadero o F para Falso).
  
  1. Boots are for cold weather. \answerline[1cm] \\
  2. You wear glasses on your feet. \answerline[1cm] \\
  3. A T-shirt is a type of clothing. \answerline[1cm] \\
  4. You wear socks under your shoes. \answerline[1cm] \\
  5. A hat goes on your head. \answerline[1cm] 
\end{activity}
