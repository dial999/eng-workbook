\section*{Session 24: Grand Final: Graduation Showcase}
\textbf{Objective:} Final presentation and course celebration.

\subsection*{The Grand Finale!}

\begin{activity}{Final Presentation Script \faIcon{star}}
  Complete this script for your presentation (Completa este guion para tu presentación).\\
  
  Hello! My name is \answerline[4cm]. I am \answerline[2cm] years old.\\
  Today, I want to present my superhero!\\
  Their name is \answerline[4cm].\\
  They live in \answerline[4cm] (a place in the city).\\
  They are wearing \answerline[5cm] (clothes and colors).\\
  They can \answerline[5cm] (superpower).\\
  They like to eat \answerline[4cm] (food).\\
  Thank you for listening!
\end{activity}

\begin{activity}{Oral Presentation Checklist \faIcon{check}}
  Before you present, check these items (Antes de presentar, revisa estos puntos):\\
  
  1. I have my script ready. $\square$\\
  2. I know how to pronounce the words. $\square$\\
  3. I am going to speak loudly and clearly. $\square$\\
  4. I have my superhero drawing. $\square$\\
  5. I am going to smile! $\square$\\
  6. I can make eye contact. $\square$\\
  7. I know my superhero's superpower. $\square$\\
  8. I know what clothes they are wearing. $\square$\\
  9. I practiced at home. $\square$\\
  10. I am ready to have fun! $\square$
\end{activity}

\begin{activity}{Peer Feedback \faIcon{smile}}
  Listen to a classmate and complete this form (Escucha a un compañero y completa este formato):\\
  
  My friend \answerline[4cm] presented their superhero.\\
  I liked \answerline[6cm].
\end{activity}

\begin{activity}{Course Review: What I Learned \faIcon{lightbulb}}
  Complete one sentence for each module (Completa una oración por cada módulo).\\
  
  Module 1: I learned how to say \answerline[5cm].\\
  Module 2: I learned the words for \answerline[5cm].\\
  Module 3: I can talk about \answerline[5cm].\\
  Module 4: I can describe \answerline[5cm].
\end{activity}
