\section*{Session 22: Prepositions of Place}
\textbf{Objective:} Locate objects using basic prepositions.

\subsection*{Vocabulary}
\begin{vocabcard}{Prepositions of Place (Preposiciones de lugar)}
  \vocabpair{IN}{Dentro de}
  \vocabpair{ON}{Encima de}
  \vocabpair{UNDER}{Debajo de}
  \vocabpair{NEXT TO}{Al lado de}
  \vocabpair{BEHIND}{Detrás de}
  \vocabpair{IN FRONT OF}{Enfrente de}
\end{vocabcard}

\subsection*{Grammar}
\begin{grammartip}{Where is it?}
  We use prepositions of place to say where things are.\\
  \textbf{Examples:}\\
  The cat is ON the table. (El gato está sobre la mesa.)\\
  The ball is UNDER the chair. (La pelota está debajo de la silla.)
\end{grammartip}

\subsection*{Activities}

\begin{activity}{Activity 1: Circle the correct preposition \faIcon{pencil-alt}}
  Read and circle the correct word (Lee y encierra en un círculo la palabra correcta).\\
  
  1. The books are ( IN / ON ) the backpack.\\
  2. The dog is sleeping ( UNDER / IN FRONT OF ) the tree.\\
  3. The cinema is ( NEXT TO / ON ) the supermarket.\\
  4. The keys are ( IN / UNDER ) my pocket.\\
  5. The teacher is standing ( IN FRONT OF / IN ) the board.\\
  6. The car is ( BEHIND / ON ) the house.
\end{activity}

\begin{activity}{Activity 2: Look around!}
  Look around your room! Write 4 sentences using different prepositions (¡Mira alrededor de tu cuarto! Escribe 4 oraciones usando diferentes preposiciones).

\end{activity}
