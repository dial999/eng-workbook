% ---- Sing & Do activity ----
\begin{activity}{Sing \& Act! · ¡Canta y actúa!}
\begin{enumerate}
  \item Read the song aloud (silently first, then out loud). \\
        \textit{Lee la canción en voz alta (primero en silencio, luego en voz alta).}
  \item Underline \underline{\ \ \ \ } all the greeting words in the song. \\
        \textit{Subraya todas las palabras de saludo en la canción.}
  \item Circle \textcircled{\ } the word that means \textbf{``nombre''} in Spanish. \\
        \textit{Encierra en un círculo la palabra que significa ``nombre'' en español.}
  \item Write TWO new sentences using greetings: \\[6pt]
        \answerline[10cm] \\[10pt]
        \answerline[10cm]
\end{enumerate}
\end{activity}
