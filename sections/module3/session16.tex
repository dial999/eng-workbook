\section*{Session 16: Daily Routine Verbs}

\subsection*{Objective}
Learn main daily actions.

\begin{vocabcard}{Daily Actions}
  \vocabpair{Wake up}{Despertar}
  \vocabpair{Brush my teeth}{Cepillar mis dientes}
  \vocabpair{Take a shower}{Tomar un baño}
  \vocabpair{Eat breakfast}{Desayunar}
  \vocabpair{Go to school}{Ir a la escuela}
  \vocabpair{Do homework}{Hacer tarea}
  \vocabpair{Watch TV}{Ver televisión}
  \vocabpair{Go to sleep}{Ir a dormir}
\end{vocabcard}

\begin{grammartip}{Talking About Routines}
  Use simple present to talk about routines. Example: \textbf{I wake up at 7 o'clock.} or \textbf{Every morning I eat breakfast.}
\end{grammartip}

\begin{songbox}{This Is the Way}
  This is the way I brush my teeth \\
  Brush my teeth, brush my teeth \\
  This is the way I brush my teeth \\
  Early in the morning! \\
  \\
  This is the way I wash my face \\
  Wash my face, wash my face \\
  This is the way I wash my face \\
  Early in the morning! \\
  \\
  This is the way I eat breakfast \\
  Eat breakfast, eat breakfast \\
  This is the way I eat breakfast \\
  Early in the morning! \\
  \\
  This is the way I go to school \\
  Go to school, go to school \\
  This is the way I go to school \\
  Early in the morning!
\end{songbox}

\begin{activity}{Activity 1: Morning Order}
  Put these 6 morning actions in the correct order (1-6). (Ordena estas 6 acciones de la mañana del 1 al 6)
  \writinglines[3]
\end{activity}

\begin{activity}{Activity 2: My Morning Routine}
  Write YOUR morning routine in 4 sentences. (Escribe tu rutina de la mañana en 4 oraciones)
  \writinglines[4]
\end{activity}

\begin{activity}{Activity 3: True or False}
  Write True or False about a classmate's routine. (Escribe Verdadero o Falso sobre la rutina de un compañero)
  \writinglines[3]
\end{activity}
