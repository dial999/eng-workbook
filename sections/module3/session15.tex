\section*{Session 15: At the Restaurant}

\subsection*{Objective}
Order food politely in a simulated context.

\begin{vocabcard}{Restaurant Phrases}
  \vocabpair{What would you like?}{¿Qué le gustaría?}
  \vocabpair{Can I have a..., please?}{¿Me da un..., por favor?}
  \vocabpair{Here you are}{Aquí tiene}
  \vocabpair{Thank you}{Gracias}
  \vocabpair{You're welcome}{De nada}
  \vocabpair{Menu}{Menú}
  \vocabpair{Main dish}{Plato principal}
  \vocabpair{Dessert}{Postre}
  \vocabpair{Drink}{Bebida}
  \vocabpair{Price}{Precio}
\end{vocabcard}

\begin{grammartip}{Being Polite}
  Always use \textbf{please} when asking for food, and \textbf{thank you} when you receive it!
\end{grammartip}

\begin{activity}{Activity 1: Design YOUR Menu}
  Write 1 main dish, 1 drink, 1 dessert and make up prices. (Diseña tu menú: 1 plato fuerte, 1 bebida, 1 postre y sus precios)
  \writinglines[3]
\end{activity}

\begin{activity}{Activity 2: Complete the Dialogue}
  Fill in the blanks to complete the waiter and customer exchange. (Completa el diálogo)
  Waiter: What would you \answerline[3cm]?
  Customer: \answerline[3cm] I have a pizza, please?
  Waiter: Here you \answerline[3cm].
  Customer: \answerline[3cm] you.
\end{activity}

\begin{activity}{Activity 3: Write a Full Order}
  Write a full order using 3 sentences. (Escribe una orden completa en 3 oraciones)
  \writinglines[3]
\end{activity}
