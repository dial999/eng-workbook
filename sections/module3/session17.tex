\section*{Session 17: Days of the Week and Time}

\subsection*{Objective}
Name days and express o'clock times.

\begin{vocabcard}{Days and Time}
  \vocabpair{Monday}{Lunes}
  \vocabpair{Tuesday}{Martes}
  \vocabpair{Wednesday}{Miércoles}
  \vocabpair{Thursday}{Jueves}
  \vocabpair{Friday}{Viernes}
  \vocabpair{Saturday}{Sábado}
  \vocabpair{Sunday}{Domingo}
  \vocabpair{What time is it?}{¿Qué hora es?}
  \vocabpair{It's ... o'clock}{Son las ... en punto}
\end{vocabcard}

\begin{grammartip}{Telling the Time}
  To say the exact hour, we use \textbf{o'clock}. Example: \textbf{It's 7 o'clock.}
\end{grammartip}

\begin{songbox}{Days of the Week}
  Monday and Tuesday \\
  Wednesday and Thursday \\
  Friday and Saturday \\
  Sunday comes after! \\
  (snap snap) The days of the week! (snap snap) \\
  The days of the week! (snap snap) \\
  The days of the week, the days of the week, the days of the week! (snap snap)
\end{songbox}

\begin{activity}{Activity 1: Write the Days}
  Write the days in order. Fill in the blanks: Monday, \answerline[2cm], Wednesday, \answerline[2cm], Friday, \answerline[2cm], Sunday. (Escribe los días en orden)
\end{activity}

\begin{activity}{Activity 2: Clock Time}
  Draw clock hands and write the time for 3 different hours. (Dibuja las manecillas del reloj y escribe la hora)
  \writinglines[3]
\end{activity}

\begin{activity}{Activity 3: English Class}
  Answer: What day is your English class? What time does it start? (Responde: ¿Qué día es tu clase de inglés? ¿A qué hora empieza?)
  \writinglines[2]
\end{activity}
