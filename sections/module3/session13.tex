\begin{grammartip}{Healthy vs Fast Food}
  \textbf{Healthy} means good for you. \textbf{Fast food} is usually cooked quickly in restaurants.
\end{grammartip}

\begin{activity}{Activity 1: Healthy or Fast Food?}
  Classify these foods into Healthy or Fast Food: Pizza, Apple, Burger, Salad, Chicken, Ice cream, Water, Cake, Soup, Sandwich. (Clasifica en saludable o comida rápida)
  \writinglines[4]
\end{activity}

\begin{activity}{Activity 2: Lunch Box}
  What is in your lunch box? Write 3 foods you usually eat. (Escribe 3 alimentos que comes en tu lonchera)
  \writinglines[3]
\end{activity}
