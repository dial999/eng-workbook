\section*{Session 14: Expressing Likes and Dislikes}

\subsection*{Objective}
Express food preferences.

\begin{vocabcard}{Food Sentences}
  \vocabpair{I like}{Me gusta}
  \vocabpair{I don't like}{No me gusta}
  \vocabpair{Do you like...?}{¿Te gusta...?}
  \vocabpair{Yes, I do}{Sí, me gusta}
  \vocabpair{No, I don't}{No, no me gusta}
\end{vocabcard}

\begin{grammartip}{Expressing Preferences}
  Use \textbf{I like} for things you enjoy eating: I like pizza.\\
  Use \textbf{I don't like} for things you don't enjoy: I don't like broccoli.\\
  To ask a question: \textbf{Do you like \underline{\hspace{2cm}}?}
\end{grammartip}

\begin{activity}{Activity 1: What do you like?}
  Write 2 foods you like and 2 you don't like in full sentences. (Escribe 2 alimentos que te gustan y 2 que no)
  \writinglines[4]
\end{activity}

\begin{activity}{Activity 2: Classmate Interview}
  Interview a classmate. Ask them ``Do you like \_\_\_?'' and write their answers. (Entrevista a un compañero)
  \writinglines[4]
\end{activity}

\begin{activity}{Activity 3: Crazy Menu}
  Rate these 5 weird food combinations with a thumbs up (\faIcon{thumbs-up}) or down (\faIcon{thumbs-down}) and write why. (Califica estas combinaciones locas y explica por qué)
  \writinglines[5]
\end{activity}
