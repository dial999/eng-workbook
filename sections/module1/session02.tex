\begin{grammartip}{\faIcon{lightbulb} \ How do you feel?}
  \textbf{How are you today?} (¿Cómo estás hoy?)\\
  \textbf{Today I feel...} (Hoy me siento...)\\[1em]
  \emph{Example: Today I feel happy!}
\end{grammartip}

\begin{activity}{\faIcon{pencil-alt} \ Activity 1: Match the Emotions (Une las emociones)}
  Match the emotion words to what they mean. (Une las palabras con su significado.)\\
  1. Happy \qquad A. Zzzzz...\\
  2. Sad \qquad B. Yay!\\
  3. Angry \qquad C. Grrr!\\
  4. Tired \qquad D. Boo hoo...
  \writinglines[4]
\end{activity}

\begin{activity}{\faIcon{check} \ Activity 2: How do you feel when...? (¿Cómo te sientes cuando...?)}
  Choose the emotion for each situation. (Elige la emoción para cada situación.)\\[1em]
  1. It is 11:00 PM.\\
  \textbf{I feel:} \answerline[4cm]\\[1em]
  2. You get a new toy.\\
  \textbf{I feel:} \answerline[4cm]\\[1em]
  3. You can't find your homework.\\
  \textbf{I feel:} \answerline[4cm]
\end{activity}

\begin{activity}{\faIcon{pencil-alt} \ Activity 3: Draw Your Face (Dibuja tu rostro)}
  Draw your face when you feel excited and when you feel sad! (¡Dibuja tu cara cuando estás emocionado y cuando estás triste!)\\[1em]
  \textbf{Excited:} \hspace{5cm} \textbf{Sad:}\\
  \vspace{3cm}
\end{activity}
