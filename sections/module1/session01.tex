\begin{grammartip}{\faIcon{lightbulb} \ Introducing Yourself}
  To ask someone's name:\\
  \textbf{What's your name?} (¿Cómo te llamas?)\\
  \textbf{My name is...} (Mi nombre es...)\\[1em]
  To ask someone's age:\\
  \textbf{How old are you?} (¿Cuántos años tienes?)\\
  \textbf{I am ... years old.} (Tengo ... años.)
\end{grammartip}

\begin{activity}{\faIcon{pencil-alt} \ Activity 1: My ID Card (Mi tarjeta de datos)}
  Fill in your personal data! (¡Completa tus datos personales!)\\[1em]
  \textbf{Hello! My name is:} \answerline[5cm]\\[1em]
  \textbf{I am} \answerline[2cm] \textbf{years old.}\\[1em]
  \emph{Draw yourself here:}\\[1em]
  \vspace{3cm}
\end{activity}

\begin{activity}{\faIcon{check} \ Activity 2: Match the Greetings (Une los saludos)}
  Match the greetings to the correct time of day or situation. (Une los saludos con la situación.)\\
  1. Good morning \qquad A. When you leave\\
  2. Good afternoon \qquad B. When you wake up\\
  3. Goodbye \qquad C. After lunch
  \writinglines[3]
\end{activity}

\begin{activity}{\faIcon{pencil-alt} \ Activity 3: Introduce Yourself (Preséntate)}
  Write 3 sentences introducing yourself. (Escribe 3 oraciones presentándote.)
  \writinglines[3]
\end{activity}
