\begin{grammartip}{\faIcon{lightbulb} \ This is vs. These are}
  Use \textbf{This is} for one object (singular):\\
  \emph{This is a pencil.} (Este es un lápiz.)\\[1em]
  Use \textbf{These are} for two or more objects (plural):\\
  \emph{These are rulers.} (Estas son reglas.)
\end{grammartip}
